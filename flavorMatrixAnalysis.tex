\begin{frame}{Pure-parallel HDM gives a full-rank four-flavor fit}
\begin{columns}[T,onlytextwidth]
\column{0.51\textwidth}
\begin{align*}
P_{\mathrm{data}}(f\mid t) &=
\frac{W_{\mathrm{data}}(t,f)}{\sum_{f'}W_{\mathrm{data}}(t,f')},\\
R_{\mathrm{data}}(t) &= \sum_f P_{\mathrm{data}}(f\mid t)
R_{\mathrm{MC}}(t,f)\,k_f,\\
R_{\mathrm{HDM}} &= \frac{m_0-m_n-m_u}
{1-m_n/R_n-m_u/R_u}.
\end{align*}
\column{0.47\textwidth}
\begin{block}{Current choices}
\begin{itemize}
\item $|\eta_{jet}|<1.3$ and pure parallel window
\item $p_{T,Z}>30$ GeV
\item $R_n=1.00$, $R_u=0.92$
\item merged component means, never event-wise HDM
\item four reco rows, four truth columns
\item rank $4/4$, condition number 4.98
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\begin{frame}{The MC truth control does not support a constant $R_u=0.92$ yet}
\begin{columns}[c,onlytextwidth]
\column{0.58\textwidth}\includegraphics[width=0.90\linewidth]{effective_ru_mc.pdf}
\column{0.40\textwidth}
\tiny
\begin{align*}
R_u^{\rm mean}&=\frac{\langle m_u^{reco}\rangle}{\langle m_u^{gen}\rangle},\\
R_u^{\rm slope}&=\frac{\langle m_u^{reco}m_u^{gen}\rangle}
{\langle(m_u^{gen})^2\rangle}.
\end{align*}
The mean ratio is unstable when $\langle m_u^{gen}\rangle$ is small. The zero-intercept slope is stable near 0.5 but includes generator/reconstruction threshold and definition migrations; it must not replace 0.92 blindly.
\end{columns}
\end{frame}

\begin{frame}{The next output makes the UE-hole subtraction explicit}
\small
For the same lepton-cleaned jets with $p_T>15$ GeV that enter reconstructed $H_T$, store
\begin{align*}
A_{\parallel} &=-\sum_j A_j\cos\Delta\phi(j,Z),\\
u_{\rm hole} &= \frac{\rho A_{\parallel}}{p_{T,Z}},\\
m_{nu}^{\rm FSR-like} &= m_n+m_u-u_{\rm hole}.
\end{align*}
\begin{itemize}
\item The scalar $\sum A_j$, $A_\parallel$, $u_{\rm hole}$, and corrected sum are stored separately for every reference-$p_T$/flavor cell.
\item Transverse pairs use the same combined $Z+$sideband projection as their $m_u$ estimator.
\item Matched MC also stores $m_{n,u}^{reco}$, $m_{n,u}^{gen}$, their products, and generator squares, enabling flavor-resolved mean-ratio and regression-slope $R_n,R_u$ estimates.
\item This preserves freedom to replace $\rho$ or model a $p_T$-dependent effective response after production.
\end{itemize}
\end{frame}

\begin{frame}{Semileptonic heavy-flavor decays require a separate truth control}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Physical effect}
Neutrinos from semileptonic c- and b-hadron decays lower the visible DB/MPF-HDM response. Depending on the particle-level jet definition, that loss need not appear in the nominal no-neutrino truth response.
\end{block}
\column{0.50\textwidth}
\begin{block}{Next-run control}
\begin{itemize}
\item identify heavy-hadron descendant neutrinos with GenPart ancestry;
\item store neutrino $p_T$ fraction and $R_{gen+\nu}/R_{gen}$ by truth/reco cell;
\item retain an ambiguous ancestry category;
\item separate decay-neutrino physics from detector response and tagging migrations.
\item use the newly stored jet PF-muon fraction as a correlated reconstructed-level proxy, not as a replacement for ancestry truth.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\begin{frame}{Recommended path before the full 2024 campaign}
\begin{enumerate}
\item Run a crossed 2-by-2 closure: old/new tags times legacy/all-pairs response on the same selected events. This is the shortest path to localizing the gluon sign change.
\item Replace fixed flavor fractions only after a data-constrained model exists; until then show the current IPF result as conditional on MC truth marginals.
\item Validate the new projected-area UE subtraction and flavor-resolved $R_n,R_u$ slopes before changing nominal HDM constants.
\item Fit $m_n+m_u-u_{\rm hole}$ jointly across the four reference-$p_T$ choices and propagate their migration covariance.
\item Add PSWeight, JER, tagger, heavy-neutrino, direct-versus-splitting, and flavor-fraction variations; later combine Z+jet, gamma+jet, and dijet data for a Jet-Topics flavor-fraction constraint.
\end{enumerate}
